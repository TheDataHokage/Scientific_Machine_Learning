def beam_on_elastic_foundation(EI, k, p, x):
    """Compute the deflection of a beam on an elastic foundation using finite difference method."""
    N = len(x)-1
    dx = x[1] - x[0]
    
    C = EI / dx**4

    A = np.zeros((N+1, N+1))
    b = p  # p supplied

    # interior nodes i=2..N-2
    for i in range(2, N-1):
        A[i, i-2] = C
        A[i, i-1] = -4*C
        A[i, i]   = 6*C + k
        A[i, i+1] = -4*C
        A[i, i+2] = C

    # i = 1 (use ghost elimination w_{-1} = -w1 since w0=0)
    A[1, 1] = 5*C + k
    A[1, 2] = -4*C
    A[1, 3] = C

    # i = N-1 (symmetric)
    A[N-1, N-3] = C
    A[N-1, N-2] = -4*C
    A[N-1, N-1] = 5*C + k

    # BC rows
    A[0, :] = 0; A[0, 0] = 1; b[0] = 0
    A[N, :] = 0; A[N, N] = 1; b[N] = 0
    w = np.linalg.solve(A, b)
    return w

start_fd = time.time()
w = beam_on_elastic_foundation(EI, k, p, x)
end_fd = time.time()
